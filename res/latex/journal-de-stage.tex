\section*{Partie B --- Journal de stage jour par jour}

\phasebanner{Phase 1 --- Découverte et apprentissage AWS (01 -- 07 juillet 2026)}

\begin{journalday}{Mercredi 01/07/2026}{Jour 1}
\journalfield{Activités réalisées}{Prise de contact avec l'organisme d'accueil, cadrage des objectifs du stage et du projet support (\texttt{aws-iam-monitor}). Mise en place de l'environnement de travail : installation et configuration d'AWS CLI, création d'un compte de travail restreint, premiers repères dans la console AWS IAM.}
\journalfield{Objectif}{Disposer d'un environnement fonctionnel avant d'aborder l'étude des services AWS et de sécuriser l'accès au compte de travail.}
\journalfield{Technologies / outils}{AWS CLI, console AWS, AWS IAM.}
\journalfield{Résultats / acquis}{Environnement de travail opérationnel ; premières notes prises dans le dépôt personnel de ressources AWS.}
\end{journalday}

\begin{journalday}{Jeudi 02/07/2026}{Jour 2}
\journalfield{Activités réalisées}{Étude du service AWS IAM : utilisateurs, groupes, rôles, politiques (policies) managées et en ligne, mécanismes d'attachement des permissions et principe du moindre privilège.}
\journalfield{Objectif}{Acquérir les bases nécessaires à la compréhension du modèle de permissions AWS, préalable indispensable à un projet de supervision IAM.}
\journalfield{Technologies / outils}{AWS IAM, AWS CLI.}
\journalfield{Résultats / acquis}{Compréhension du modèle utilisateurs/groupes/rôles/politiques et des risques associés à une mauvaise configuration IAM.}
\end{journalday}

\begin{journalday}{Vendredi 03/07/2026}{Jour 3}
\journalfield{Activités réalisées}{Étude d'AWS CloudTrail : notion de \textit{management event}, structure d'un événement (acteur, action, ressource, adresse IP source), cas d'usage en audit et en détection de sécurité.}
\journalfield{Objectif}{Comprendre comment CloudTrail capture les appels d'API IAM, ce qui conditionne toute la chaîne de détection du futur projet.}
\journalfield{Technologies / outils}{AWS CloudTrail.}
\journalfield{Résultats / acquis}{Compréhension du format des événements CloudTrail et de leur exploitation pour la détection d'opérations sensibles.}
\end{journalday}

\begin{journalday}{Lundi 06/07/2026}{Jour 4}
\journalfield{Activités réalisées}{Étude d'Amazon EventBridge (règles, \textit{event pattern}, cibles) et d'AWS Lambda (modèle d'exécution serverless, runtimes disponibles, rôle d'exécution IAM).}
\journalfield{Objectif}{Comprendre le mécanisme de filtrage événementiel et le fonctionnement d'une fonction serverless, briques centrales de l'architecture à concevoir.}
\journalfield{Technologies / outils}{Amazon EventBridge, AWS Lambda, Python.}
\journalfield{Résultats / acquis}{Compréhension du couple EventBridge/Lambda comme mécanisme de traitement événementiel en quasi temps réel.}
\end{journalday}

\begin{journalday}{Mardi 07/07/2026}{Jour 5}
\journalfield{Activités réalisées}{Étude d'Amazon SNS (topics, abonnements), Amazon S3 (chiffrement, versioning) et Amazon CloudWatch (logs, métriques personnalisées, alarmes). Synthèse de la semaine et cadrage définitif du sujet de projet : plateforme serverless de supervision et d'audit des changements IAM.}
\journalfield{Objectif}{Clore la phase de découverte AWS en couvrant l'ensemble des services identifiés comme nécessaires, puis formaliser le périmètre du projet à développer.}
\journalfield{Technologies / outils}{Amazon SNS, Amazon S3, Amazon CloudWatch.}
\journalfield{Résultats / acquis}{Vision d'ensemble des services AWS mobilisés par le projet ; sujet de stage formalisé autour du dépôt \texttt{aws-iam-monitor}.}
\journalfield{Observations}{Cette semaine a été consacrée exclusivement à l'apprentissage ; aucun développement n'a encore été entamé sur le projet.}
\end{journalday}

\phasebanner{Phase 2 --- Développement du projet aws-iam-monitor (08 juillet -- 21 août 2026)}

\journalfigure{aws-iam-monitor-architecture-diagram.png}{0.92}{Architecture cible du projet \texttt{aws-iam-monitor}, retenue à l'issue de la phase de conception (Jour 8) et provisionnée à l'identique par les deux implémentations développées durant cette phase.}

\begin{journalday}{Mercredi 08/07/2026}{Jour 6}
\journalfield{Activités réalisées}{Analyse du besoin de sécurité : identification des opérations IAM sensibles à surveiller en priorité (\texttt{CreateUser}, \texttt{DeletePolicy}, \texttt{CreateAccessKey}, \texttt{DeactivateMFADevice}, \texttt{AttachUserPolicy}) et des scénarios de menace associés (création de compte non autorisée, désactivation de MFA, escalade de privilèges).}
\journalfield{Objectif}{Cadrer précisément ce que le système doit détecter avant de concevoir l'architecture technique.}
\journalfield{Technologies / outils}{AWS IAM, AWS CloudTrail.}
\journalfield{Résultats / acquis}{Liste des opérations IAM jugées critiques pour la supervision, base de la future logique de classification des risques.}
\end{journalday}

\begin{journalday}{Jeudi 09/07/2026}{Jour 7}
\journalfield{Activités réalisées}{Étude approfondie de la manière dont CloudTrail journalise les appels d'API IAM au niveau compte, et réflexion sur le filtrage de ces événements par Amazon EventBridge (event pattern ciblant la source \texttt{iam.amazonaws.com}).}
\journalfield{Objectif}{Préparer la conception du flux de données qui reliera CloudTrail à la couche de traitement.}
\journalfield{Technologies / outils}{AWS CloudTrail, Amazon EventBridge.}
\journalfield{Résultats / acquis}{Principe de filtrage retenu : un \textit{event pattern} EventBridge large sur la source IAM, le filtrage fin des actions surveillées étant délégué à la logique applicative de la Lambda.}
\end{journalday}

\begin{journalday}{Vendredi 10/07/2026}{Jour 8}
\journalfield{Activités réalisées}{Conception du flux général de l'architecture : IAM $\rightarrow$ CloudTrail $\rightarrow$ EventBridge $\rightarrow$ Lambda $\rightarrow$ (SNS + S3 + CloudWatch). Choix motivé des services AWS pour chaque étage (capture, filtrage, traitement, notification, archivage, supervision).}
\journalfield{Objectif}{Fixer une architecture serverless cohérente avant tout développement.}
\journalfield{Technologies / outils}{AWS CloudTrail, Amazon EventBridge, AWS Lambda, Amazon SNS, Amazon S3, Amazon CloudWatch.}
\journalfield{Résultats / acquis}{Schéma d'architecture validé, qui servira de référence pour les deux implémentations (Bash et Terraform).}
\end{journalday}

\begin{journalday}{Lundi 13/07/2026}{Jour 9}
\journalfield{Activités réalisées}{Conception de la stratégie de classification des risques (niveaux \texttt{CRITICAL}, \texttt{HIGH}, \texttt{MEDIUM}, \texttt{LOW}) et de la logique de \textit{whitelist} d'adresses IP, destinée à qualifier une action IAM comme provenant ou non d'une source de confiance.}
\journalfield{Objectif}{Définir les règles métier de la fonction Lambda avant son implémentation.}
\journalfield{Technologies / outils}{AWS Lambda (conception), AWS IAM.}
\journalfield{Résultats / acquis}{Grille de classification des risques et principe de génération dynamique d'une politique \texttt{IPWhitelistPolicy}.}
\end{journalday}

\begin{journalday}{Mardi 14/07/2026}{Jour 10}
\journalfield{Activités réalisées}{Démarrage de l'implémentation Bash/AWS CLI : mise en place de la structure du répertoire \texttt{project/bash/} (bibliothèque partagée \texttt{lib/} avec les fonctions communes, le module de journalisation et la vérification des prérequis) et squelette du script d'orchestration \texttt{awsctl}.}
\journalfield{Objectif}{Poser une base modulaire et réutilisable avant d'implémenter chaque composant AWS.}
\journalfield{Technologies / outils}{Bash, AWS CLI.}
\journalfield{Résultats / acquis}{Squelette fonctionnel de \texttt{awsctl} et bibliothèque d'aide commune aux différents modules.}
\end{journalday}

\begin{journalday}{Mercredi 15/07/2026}{Jour 11}
\journalfield{Activités réalisées}{Développement du module IAM (Bash) : création des politiques gérées \texttt{DevPolicy}, \texttt{ProdPolicy} et \texttt{AuditPolicy}, ainsi que des groupes \texttt{DevGroup}, \texttt{ProdGroup} et \texttt{TestGroup}.}
\journalfield{Objectif}{Mettre en place le bac à sable IAM de démonstration nécessaire au fonctionnement du système.}
\journalfield{Technologies / outils}{AWS IAM, AWS CLI, Bash.}
\journalfield{Résultats / acquis}{Politiques et groupes IAM créés de façon scriptée et idempotente.}
\end{journalday}

\begin{journalday}{Jeudi 16/07/2026}{Jour 12}
\journalfield{Activités réalisées}{Poursuite du module IAM (Bash) : création des utilisateurs de démonstration (\texttt{DevAdmin}, \texttt{ProdAdmin}, etc.), ajout de l'option \texttt{-r}/\texttt{--role} pour la création optionnelle de rôles IAM, et génération dynamique de la politique \texttt{IPWhitelistPolicy} à partir de la variable \texttt{ALLOWED\_IPS}.}
\journalfield{Objectif}{Compléter le module IAM avec la gestion des utilisateurs et la logique de restriction par adresse IP.}
\journalfield{Technologies / outils}{AWS IAM, Bash.}
\journalfield{Résultats / acquis}{Module IAM complet, incluant la génération conditionnelle de rôles et de la politique de whitelist.}
\end{journalday}

\begin{journalday}{Vendredi 17/07/2026}{Jour 13}
\journalfield{Activités réalisées}{Développement du module CloudTrail (Bash) : création du bucket S3 destiné aux journaux (chiffrement AES-256, versioning activé) puis du \textit{trail} CloudTrail capturant les événements de management sur l'ensemble du compte.}
\journalfield{Objectif}{Mettre en place la source d'événements sur laquelle repose toute la chaîne de détection.}
\journalfield{Technologies / outils}{AWS CloudTrail, Amazon S3, Bash.}
\journalfield{Résultats / acquis}{Trail CloudTrail opérationnel, journalisant les appels d'API dans un bucket S3 sécurisé.}
\end{journalday}

\begin{journalday}{Lundi 20/07/2026}{Jour 14}
\journalfield{Activités réalisées}{Développement du module SNS (Bash) : création du topic d'alerte \texttt{iam-alerts}, définition de sa politique d'accès, et abonnement automatique d'une adresse e-mail si la variable \texttt{SNS\_ALERT\_EMAIL} est renseignée dans le fichier \texttt{.env}.}
\journalfield{Objectif}{Mettre en place le canal de notification des événements à haut risque.}
\journalfield{Technologies / outils}{Amazon SNS, Bash.}
\journalfield{Résultats / acquis}{Topic SNS fonctionnel avec abonnement e-mail configurable.}
\end{journalday}

\begin{journalday}{Mardi 21/07/2026}{Jour 15}
\journalfield{Activités réalisées}{Démarrage du module Lambda (Bash), première partie : création du bucket S3 d'historique d'audit, du rôle d'exécution de la fonction, et packaging initial du code \texttt{lambda\_function.py} (Python 3.11).}
\journalfield{Objectif}{Préparer l'infrastructure d'exécution avant d'implémenter la logique métier de traitement des événements.}
\journalfield{Technologies / outils}{AWS Lambda, Python 3.11, Amazon S3, AWS IAM, Bash.}
\journalfield{Résultats / acquis}{Rôle d'exécution et bucket d'audit provisionnés ; premier déploiement du squelette de la fonction Lambda.}

\begin{journalday}{Mercredi 22/07/2026}{Jour 16}
\journalfield{Activités réalisées}{Implémentation de la logique de traitement du payload CloudTrail reçu par la fonction Lambda, et de la vérification de l'adresse IP source de l'appel d'API par rapport à la politique de whitelist définie précédemment.}
\journalfield{Objectif}{Permettre à la fonction de distinguer une opération IAM légitime d'une opération provenant d'une source non autorisée.}
\journalfield{Technologies / outils}{Python, AWS Lambda.}
\journalfield{Résultats / acquis}{Fonction capable d'extraire les champs pertinents de l'événement et de qualifier la source de l'appel.}

\begin{journalday}{Jeudi 23/07/2026}{Jour 17}
